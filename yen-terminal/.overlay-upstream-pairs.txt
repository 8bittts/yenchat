# Overlays that REPLACE an upstream file: overlay-relative path, TAB, and the
# SHA256 of UPSTREAM's copy at the pinned commit.
#
# [3a/5] in scripts/10-update-ghostty.sh walks this against the freshly
# downloaded PRISTINE tree and fails closed on either failure mode:
#   path missing  -> upstream moved or deleted a file YEN forks. Without this
#                    the stale fork is resurrected with no signal.
#   hash changed  -> upstream EDITED a file YEN overwrites wholesale. The
#                    overlay silently discards that edit; there is no diff and
#                    no compile error. This is exactly how upstream's
#                    normalizeArchive work in src/build/LibtoolStep.zig was
#                    being thrown away on every bump, and how the entire Find
#                    menu was lost from macos/Sources/App/MainMenu.xib.
#
# Paths NOT listed are YEN-only files and are not checked.
#
# REGENERATE AGAINST PRISTINE UPSTREAM, NEVER AGAINST yen-terminal/ghostty/ —
# that directory has overlays applied, so every YEN-only file looks like an
# upstream pair. An earlier version built that way was 225/265 wrong.
#
#   PIN=$(grep '^upstream_commit=' yen-terminal/.ghostty-baseline | cut -d= -f2)
#   curl -sL "https://github.com/ghostty-org/ghostty/archive/${PIN}.tar.gz" | tar xz -C /tmp
#   cd yen-terminal && /usr/bin/find overlays -type f -not -path '*/.zig-cache/*' \
#     | while read -r f; do rel="${f#overlays/}"; u="/tmp/ghostty-${PIN}/$rel"; \
#         [ -f "$u" ] && printf '%s\t%s\n' "$rel" "$(shasum -a 256 "$u" | awk '{print $1}')"; done \
#     | LC_ALL=C sort > .overlay-upstream-pairs.txt
#
# Re-pin only after reading upstream's change and porting what applies.
